Android Picker 组件适配

2026/11/7 编程 Android

场景梳理

适配指南：https://dev.mi.com/xiaomihyperos/documentation/detail?pId=2224

涉及的场景照片视频、文件、剪贴板、日历（联系人未涉及）需要进行统一的 Pick 组件适配。

当前支持 Picker 的操作系统版本如下：

  • HyperOS 3（基于 Android 16）及以上
  • ColorOS 16（基于 Android 16）及以上
  • OriginOS 6（基于 Android 16）及以上
  • MagicOS 10（基于 Android 16）及以上

SystemPhotoPickerHelper 工具类

package com.lofter.android.album.systemUtils

import android.app.Activity
import android.content.ClipData
import android.content.Intent
import android.net.Uri
import android.provider.MediaStore
import lofter.framework.log.wmlog.NTLog
import lofter.framework.tools.utils.business.AndroidUtil

/**
 * Android 16+ 系统 Photo Picker 适配工具类
 */
object SystemPhotoPickerHelper {

    private const val TAG = "SystemPhotoPickerHelper"

    const val SYSTEM_PICKER_REQUEST_CODE = 10035

    private const val EXTRA_PICKER_PRE_SELECTION_URIS = "android.provider.extra.PICKER_PRE_SELECTION_URIS"

    /**
     * 是否可以使用系统 Photo Picker
     *
     * 1. Android 16 (API 36)
     * 2. 设备支持 ACTION_PICK_IMAGES
     */
    @JvmStatic
    fun shouldUseSystemPicker(activity: Activity?): Boolean {
        if (activity == null || !AndroidUtil.isCheckSystemVersion()) {
            return false
        }

        return try {
            val intent = Intent(MediaStore.ACTION_PICK_IMAGES)
            activity.packageManager.resolveActivity(intent, 0) != null
        } catch (e: Exception) {
            NTLog.e(TAG, "shouldUseSystemPicker error: ${e.message}")
            false
        }
    }

    /**
     * 判断是否为“部分访问”权限状态（Android 16+ 特性）
     *
     * 当用户在权限弹窗中选择“选择部分照片”时，
     * 应该引导使用系统 Photo Picker 以获取更好的体验
     */
    @JvmStatic
    fun isPartialAccessGranted(activity: Activity?): Boolean {
        if (activity == null || !AndroidUtil.isCheckSystemVersion()) {
            return false
        }

        return try {
            // READ_MEDIA_VISUAL_USER_SELECTED 是 Android 14 新增的权限
            // 表示用户选择了“部分访问”
            val permission = "android.permission.READ_MEDIA_VISUAL_USER_SELECTED"
            activity.checkSelfPermission(permission) == android.content.pm.PackageManager.PERMISSION_GRANTED
        } catch (e: Exception) {
            NTLog.e(TAG, "isPartialAccessGranted error: ${e.message}")
            false
        }
    }

    /**
     * 启动系统 Photo Picker（单选模式）
     *
     * @param activity 当前 Activity
     * @param onlyImage true: 仅图片, false: 图片和视频
     */
    @JvmStatic
    fun launchPickerSingle(activity: Activity, onlyImage: Boolean = true) {
        if (AndroidUtil.isCheckSystemVersion()) {
            val intent = Intent(MediaStore.ACTION_PICK_IMAGES).apply {
                if (onlyImage) {
                    type = "image/*"
                }
            }

            NTLog.d(TAG, "launchPickerSingle, onlyImage: $onlyImage")
            activity.startActivityForResult(intent, SYSTEM_PICKER_REQUEST_CODE)
        }
    }

    /**
     * 启动系统 Photo Picker（多选模式）
     *
     * @param activity 当前 Activity
     * @param maxCount 最大选择数量 (1-100)
     * @param onlyImage true: 仅图片, false: 图片和视频
     * @param preSelectedUris 预选的 Uri 列表（可选）
     */
    @JvmStatic
    @JvmOverloads
    fun launchPickerMultiple(
        activity: Activity,
        maxCount: Int = 9,
        onlyImage: Boolean = false,
        preSelectedUris: ArrayList<Uri>? = null
    ) {
        if (AndroidUtil.isCheckSystemVersion()) {
            val safeMaxCount = maxCount.coerceIn(1, 100)

            val intent = Intent(MediaStore.ACTION_PICK_IMAGES).apply {
                if (onlyImage) {
                    type = "image/*"
                }
                putExtra(MediaStore.EXTRA_PICK_IMAGES_MAX, safeMaxCount)

                // 预选设置（Android 14+）
                if (!preSelectedUris.isNullOrEmpty()) {
                    putExtra(EXTRA_PICKER_PRE_SELECTION_URIS, preSelectedUris)
                }
            }

            NTLog.d(TAG, "launchPickerMultiple, maxCount: $safeMaxCount, onlyImage: $onlyImage")
            activity.startActivityForResult(intent, SYSTEM_PICKER_REQUEST_CODE)
        }
    }

    /**
     * 处理系统 Photo Picker 返回结果
     *
     * @param requestCode 请求码
     * @param resultCode 结果码
     * @param data Intent 数据
     * @return 选中的 Uri 列表，如果取消或失败返回 null
     */
    @JvmStatic
    fun handlePickerResult(
        requestCode: Int,
        resultCode: Int,
        data: Intent?
    ): ArrayList<Uri>? {
        if (requestCode != SYSTEM_PICKER_REQUEST_CODE) {
            return null
        }

        if (resultCode != Activity.RESULT_OK || data == null) {
            NTLog.d(TAG, "handlePickerResult: cancelled or no data")
            return null
        }

        val resultUris = ArrayList<Uri>()

        val clipData: ClipData? = data.clipData
        if (clipData != null && clipData.itemCount > 0) {
            for (i in 0 until clipData.itemCount) {
                clipData.getItemAt(i)?.uri?.let { uri ->
                    resultUris.add(uri)
                }
            }
            NTLog.d(TAG, "handlePickerResult: multi-select, count: ${resultUris.size}")
        } else {
            data.data?.let { uri ->
                resultUris.add(uri)
                NTLog.d(TAG, "handlePickerResult: single-select, uri: $uri")
            }
        }

        return resultUris.ifEmpty {
            null
        }
    }

    @JvmStatic
    fun checkRequestCode(requestCode: Int): Boolean {
        return requestCode == SYSTEM_PICKER_REQUEST_CODE
    }
}
