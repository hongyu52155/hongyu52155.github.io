Flutter 多引擎改造实践

2026/11/8 编程 Flutter

前言

通过 FlutterBoost 实现的单引擎，现在迁移到多引擎就无法继续直接使用 FlutterBoost（主要是也没这个必要继续使用 FlutterBoost，所以就不考虑改造 FlutterBoost 了）。但是为了减少原生工程的改动，改造后还是尽量贴合原 FlutterBoost 的调用逻辑。

为什么要换多引擎？

针对多引擎和单引擎之间的优缺点，再结合我们项目的实际情况，自己取舍是否有必要使用多引擎。

单引擎 多引擎
内存释放 ❌ 内存会常驻，不会跟随页面销毁而销毁，但是优点是在 App 应用生命周期内只会有一份引擎内存开销 ✅ 内存随用随销，不会因为开启的页面多了而导致应用内存持续居高不下，但是缺点是跟随每个页面（局部组件）创建而 spawn 一份新的引擎
同屏多视图 ❌ 同屏幕中只能存在一个 Flutter 视图，不管是页面还是局部视图 ✅ 可以在同一个屏幕中绘制多个不同的 Flutter 视图
插件适配 ✅ 不存在插件适配问题 ❌ 少部分插件可能在多引擎环境下无法使用，需要针对多引擎重新适配（如 PowerImage）
热刷/热更调试 ✅ 不存在调试问题，Flutter Attach 可直接调用 ❌ 因多引擎会产生多个入口，在 attach 时无法预知需调试哪个入口，只能通过热刷后重新初始化页面/组件进行调试
通信通道 ✅ 因为只有一个引擎，所以只需要有一份 method_channel ❌ 每个引擎都要创建一个新的 channel，并统一管理

在我们项目中，考虑到未来可能会出现原生页面中使用局部 Flutter 视图，且单引擎下很容易内存持续维持在高位（因为我们有些 Flutter 页面中有很多图片或者 lottie 动画，会导致内存暴涨）。结合以上两点问题，所以我们决定将单引擎更换到多引擎上。

class LoftContainer extends InheritedWidget {
  final Map<String, dynamic>? arg;

  const LoftContainer({required Widget child, Key? key, this.arg})
      : super(key: key, child: child);

  static LoftContainer? of(BuildContext context) {
    if (context.dependOnInheritedWidgetOfExactType<LoftContainer>() != null) {
      return context.dependOnInheritedWidgetOfExactType<LoftContainer>()
          as LoftContainer;
    } else {
      return null;
    }
  }

  @override
  bool updateShouldNotify(covariant LoftContainer oldWidget) {
    return arg.toString() != oldWidget.arg.toString();
  }
}

    // 一般原生都会实现这种 schema 跳转或者应用内路由跳转的 api

// 3. 实现从 iOS 原生页面跳转到 Flutter 页面
func pushFlutterRoute(_ options: FlutterRouteOptions!) {
    guard let pageName = options.pageName else { return }

    let letters = "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789"
    let uniqueId = "\(String((0..<32).map { _ in letters.randomElement()! }))-\(isInWindow ? "1" : "0")"

    let flutterVC = LFTFlutterViewController()
    flutterVC.configFlutter(pageName: pageName, uniqueId: uniqueId, params: options.arguments, opaque: options.opaque)

    let isPresent = (options.arguments?["isPresent"] as? Bool) ?? false
    let isAnimated = (options.arguments?["isAnimated"] as? Bool) ?? true

    // 保存 Flutter 页面返回原生页面时的回调
    // 避免同时打开多个相同 pageName 的 Flutter 视图导致回调异常
    // 这里使用 uniqueId 而不是 pageName 作为标识
    resultTable[uniqueId] = options.onPageFinished

    if isPresent || !options.opaque {
        let pnav = PanNavigationController(rootViewController: flutterVC)
        pnav.modalPresentationStyle = .overFullScreen
        pnav.modalPresentationCapturesStatusBarAppearance = true
        if let vc = sharedAppDelegate().topPresentedViewController() {
            vc.present(pnav, animated: isAnimated)
        }
    } else {
        guard let panNavigationController = Util.findPushViewController()?.panNavigationController else { return }
        panNavigationController.pushViewController(flutterVC, hasTitleBar: false)
    }
}

// 4. 实现从 Flutter 页面返回原生页面
func popRoute(_ arguments: [AnyHashable: Any]) {
    let flutterVC = Util.findPushViewController() as? LFTFlutterViewController
    guard let pageInfo = arguments["pageInfo"] as? [AnyHashable: Any] else { return }
    guard let uniqueId = pageInfo["uniqueId"] as? String else { return }

    if let vc = flutterVC, vc.uniqueIdString() == uniqueId, vc.presentingViewController != nil { // dismiss
        popRouteWhenNeedDismissVC(vc, arguments: arguments)
    } else { // pop
        popRouteWhenNeedPopVC(arguments)
        configPageFinshed(arguments: arguments)
    }
}

// 5. 实现 Flutter 页面消失后，触发回调，同时移除该 uniqueId 对应的回调
private func configPageFinshed(arguments: [AnyHashable: Any]) {
    let pageInfo = arguments["pageInfo"] as? [AnyHashable: Any] ?? [:]
    guard let uniqueId = pageInfo["uniqueId"] as? String else { return }
    guard let onPageFinshed = resultTable[uniqueId] else { return }

    onPageFinshed(arguments)
    resultTable.removeValue(forKey: uniqueId)
}

5. 原生主动发送通信至 Flutter

/// 从原生模块传递信息给 Flutter 模块，以此来调用 Flutter 模块的方法
class func invokeFlutterMethod(moduleName: String, methodName: String, arguments: Dictionary<String, Any>?, completion: (() -> Void)?) {
    let methodNameWithModule = moduleName + "_" + methodName
    for channel in LFTRegisterFlutter.methodChannel {
        channel.invokeMethod(methodNameWithModule, arguments: arguments) { result in
            completion?()
        }
    }
}

5. 原生主动发送通信至 Flutter

/// 从原生模块传递信息给 Flutter 模块，以此来调用 Flutter 模块的方法
fun invokeFlutterMethod(moduleName: String, methodName: String, arguments: Map<String, Any?>, callback: FlutterNativeMethodChannelResult?) {
    for (channel in methodChannels) {
        channel.invokeMethod("${moduleName.toLowerCase()}_$methodName", arguments, object : MethodChannel.Result {
            override fun success(result: Any?) {
                callback?.success(result)
            }

            override fun error(errorCode: String, errorMessage: String?, errorDetails: Any?) {
                callback?.error(errorCode, errorMessage, errorDetails)
            }

            override fun notImplemented() {
                callback?.notImplemented()
            }
        })
    }
}

引擎优化

在多引擎环境下，发现页面跳转时有明显的视图创建过程（即页面从空白到渲染出对应 Flutter 页面）。同时当页面销毁时，因为对应引擎被销毁，将会跳过 Flutter 组件的 dispose 调用（类似于应用从后台直接杀死将不会走完完整的生命周期）。基于目前发现的两点问题，做了以下优化。

优化点 1：优化视图渲染耗时

测试时发现进入 Flutter 页面从创建引擎到渲染视图中间需要 250~300ms 的耗时，单单创建引擎就需要占用 100ms 多的时间。遂突发奇想：如果我们将「初始化引擎」这一步的时间提前——意味着当我们首次创建引擎（预热引擎）时，就要创建两个引擎，同时为了避免引擎被释放，我们要将第二个创建的引擎持有。而后进入 FlutterVC 时，读取这个已被创建但未使用的引擎，再在下一个 runloop 中创建下一个空置引擎（避免因为在当前线程中创建引擎而占用视图渲染时间）。

func warmUpEngine() {

修改创建引擎代码：

@discardableResult func makeNewEngine(isInitial: Bool = false) -> LFTFlutterEngineModel? {
    let letters = "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789"
    let uniqueIdStr = "\(String((0..<32).map { _ in letters.randomElement()! }))"

    let newEngine = LFTRegisterFlutter.groupEngine.makeEngine(withEntrypoint: nil, libraryURI: nil, initialRoute: isInitial ? "initialRoute" : uniqueIdStr)
    if !isInitial {
        let model = LFTFlutterEngineModel()
        GeneratedPluginRegistrant.register(with: newEngine)
        model.engine = newEngine
        model.uniqueId = uniqueIdStr
        engines.append(model)
        return model
    } else {
        self.initializedEngine = newEngine
        return nil
    }
}

获取空置引擎：

func getLastUnuseEngine() -> LFTFlutterEngineModel? {
    for model in engines {
        if model.bindVC == nil {
            return model
        }
    }
    return nil
}

销毁引擎：

func destroyEngine(model: LFTFlutterEngineModel?) {
    guard let model = model else { return }
    // 销毁引擎
    model.engine?.destroyContext()
    // 将引擎从数组中移除
    engines.removeAll { element in
        return element == model
    }
}

修改 FlutterViewController 的创建流程：

private func createEngine() {
    // 获取上次创建好的空置引擎
    if let engineModel = LFTRegisterFlutter.instance.getLastUnuseEngine(),
       let engine = engineModel.engine {
        self.engineModel = engineModel
        self.flutterVC = FlutterViewController(engine: engine, nibName: nil, bundle: nil)
        // 赋值当前引擎所绑定的 VC
        engineModel.bindVC = self.flutterVC
        self.addMethodChannel(engine)
        self.addFlutterView(false)
    }
    // 在下一次 runloop 中创建下一个引擎
    DispatchQueue.main.async {
        LFTRegisterFlutter.instance.makeNewEngine()
    }
}

测试结果

修改前：进入 Flutter 页面从创建引擎到渲染视图需要 250~300ms；修改后：平均创建到渲染耗时 150ms 左右，有时候甚至只有 30ms 的耗时。

优化点 2：视图销毁时机

关于销毁时机，我最初的想法是当视图即将被销毁时，通知 Flutter 页面销毁视图，并且强行持有当前 VC，延迟一秒后再释放引擎（为了保证此时已触发了完整的 dispose）。然后同事反馈，这种做法属实有点 low。

为什么当时选择用延迟一秒的方式去释放呢？主要原因在于，Flutter 中的 dispose 是生命周期最后一个方法，其中也没有是否完成 dispose 的判断方式。

后来有一天突然想到，Flutter 的 eventLoop 是优先处理主任务，其次是 MicroTask，最后才会继续处理 eventQueue。而 dispose 的任务队列处于第一梯队的主任务中，那我们就可以利用异步任务 Future 降低任务执行优先级（Future 任务在 eventQueue 队列中执行），以此达到当 dispose 方法执行完毕后再通过通信通道告知原生侧，视图已被完全销毁。

改造释放回调核心

首先第一点，就是需要在 FlutterViewController 即将被销毁时通知到 Flutter 侧，并且原生侧持有该 VC 延迟释放。我的办法就是利用 GCD 中的延迟调用，达到强行持有当前 VC 的目的：

// 这里还有一点需要注意，不是所有 disappear 都表示视图即将被销毁
// 也有可能是视图即将 push 或者 present，需要自行判断
override func viewDidDisappear(_ animated: Bool) {
    // 当视图即将被销毁时，通信通道告知 Flutter 页面移除当前组件
    channel?.invokeMethod("view_viewDispose", arguments: nil)
    // 通过 DispatchWorkItem 回调中持有当前对象 self，这里的 description 只是为了持有
    workItem = DispatchWorkItem(block: {
        _ = self.description
    })
    // 避免特殊情况下未触发销毁回调，所以延迟 5 秒做个保护，5 秒后强行释放
    DispatchQueue.main.asyncAfter(deadline: .now() + 5) { [weak self] in
        self?.workItem?.cancel()
        self?.workItem = nil
    }
}

通过 method_channel 回调，销毁当前强行持有的 workItem：

channel?.setMethodCallHandler({ [weak self] call, reply in
    let arguments = call.arguments as? Dictionary<String, Any>
    if call.method == "view_dispose" {
        self?.workItem?.cancel()
        self?.workItem = nil
    } else {
        // xxxxxxx.....
    }
})

Flutter 侧新增监听：

static void invokeMethod(String methodName, Map? arguments) {
  final methodNameType = methodName.toEnum(LFTFlutterViewMethodName.values);
  switch (methodNameType) {
    case LFTFlutterViewMethodName.viewDispose:
      // 当收到视图即将被销毁的通知时，移除当前视图
      mainWidgetKey.currentState?.remove();
    default:
  }
}

并且修改 dispose 方法：

@override
void dispose() {
  super.dispose();
  // 通过 Future 降低任务队列优先级，保证视图被 dispose 完成后再触发通知
  Future(() {
    debugPrint("future 释放完成");
    LFTMethodChannel.invokeNativeMethod("view_dispose", null);
  });
}

修改后：视图销毁时 dispose 能够被完整执行，原生侧在收到 view_dispose 回调后再释放引擎，不再依赖固定延迟。
